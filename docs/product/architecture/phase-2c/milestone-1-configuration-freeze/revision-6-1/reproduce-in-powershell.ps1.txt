param([string]$SignerDir, [string]$SdkPackage, [string]$Config = '', [string]$ManifestOut = '', [string]$Verify)
# NON-AUTHORITATIVE reimplementation of the canonical baseline PROCEDURE (version 1, Revision 6) in PowerShell: no Node, no shared code.
# WHAT IT DOES: hashes a FIXED list of eight closure files (including pinned.cjs), the deployment configuration (or ABSENT), and the two pinned
#   SDK files; applies the same text validation as the Node tool (CR byte, UTF-8 BOM, invalid UTF-8 are refused; strict UTF-8 decoding);
#   writes the same manifest (header, "<sha256>  <label>", ordinal order, LF, final newline); prints the manifest and the baseline hash.
# WHAT IT DOES NOT DO: it does NOT discover the require graph and does NOT apply the loading policy (no parser here). A closure drift or a
#   forbidden loading form is detected ONLY by the Node tool. This script therefore proves manifest/hash parity for a given set of files,
#   not that the file list is complete. Exit codes: 0 ok / MATCH, 1 MISMATCH, 2 refused, 3 malformed -Verify value.
$ErrorActionPreference = 'Stop'
$closure = 'authorization.cjs','dpapi-key.cjs','pinned.cjs','refusal.cjs','sdk-loader.cjs','state-store.cjs','tst-signer-core.cjs','tst-signer.cjs'
function Sha([byte[]]$bytes) { $h = [Security.Cryptography.SHA256]::Create().ComputeHash($bytes); (($h | ForEach-Object { $_.ToString('x2') }) -join '') }
function Canonical([string]$path) {
  [byte[]]$b = [IO.File]::ReadAllBytes($path)
  foreach ($x in $b) { if ($x -eq 13) { throw "$path contains a CR byte" } }
  if ($b.Length -ge 3 -and $b[0] -eq 0xEF -and $b[1] -eq 0xBB -and $b[2] -eq 0xBF) { throw "$path starts with a UTF-8 BOM" }
  try { [void]([Text.UTF8Encoding]::new($false, $true)).GetString($b) } catch { throw "$path is not valid UTF-8" }
  Sha $b
}
try {
  if (-not $SignerDir -or -not $SdkPackage) { throw 'usage: -SignerDir <dir> -SdkPackage <dir> [-Config <file>] [-ManifestOut <file>] [-Verify <64 hex>]' }
  $d = New-Object 'System.Collections.Generic.SortedDictionary[string,string]' ([StringComparer]::Ordinal)
  foreach ($f in $closure) { $d["file:$f"] = Canonical (Join-Path $SignerDir $f) }
  if ($Config) { $d['config:deployment-config.json'] = Canonical $Config } else { $d['config:deployment-config.json'] = 'ABSENT' }
  $d['external:@stellar/stellar-sdk@17.4.0/package.json'] = Sha ([IO.File]::ReadAllBytes((Join-Path $SdkPackage 'package.json')))
  $d['external:@stellar/stellar-sdk@17.4.0/dist/stellar-sdk.js'] = Sha ([IO.File]::ReadAllBytes((Join-Path (Join-Path $SdkPackage 'dist') 'stellar-sdk.js')))
  $sb = New-Object System.Text.StringBuilder
  [void]$sb.Append("TOBIN-TST-SIGNER-BASELINE v1`n")
  foreach ($k in $d.Keys) { [void]$sb.Append($d[$k] + '  ' + $k + "`n") }
  $bytes = ([Text.UTF8Encoding]::new($false)).GetBytes($sb.ToString())
  if ($ManifestOut) { [IO.File]::WriteAllBytes($ManifestOut, $bytes) }
  $hash = Sha $bytes
  [Console]::Out.Write($sb.ToString()); [Console]::Out.WriteLine(''); [Console]::Out.WriteLine("baseline sha256: $hash")
  if ($PSBoundParameters.ContainsKey('Verify')) {
    if ($Verify -cnotmatch '^[0-9a-f]{64}$') { [Console]::Out.WriteLine('VERIFY: REFUSED (the expected value is not 64 lowercase hex characters)'); exit 3 }
    if ($Verify -ceq $hash) { [Console]::Out.WriteLine('VERIFY: MATCH'); exit 0 } else { [Console]::Out.WriteLine('VERIFY: MISMATCH'); exit 1 }
  }
  exit 0
} catch { [Console]::Error.WriteLine("REFUSED: $($_.Exception.Message)"); exit 2 }
