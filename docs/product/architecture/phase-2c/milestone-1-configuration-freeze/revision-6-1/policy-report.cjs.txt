'use strict';
// Generates globalThis-access-inventory.md from the analyzer and the signer sources. Offline; reads files only.
// Usage: node policy-report.cjs <signer-dir> > globalThis-access-inventory.md
const fs = require('fs'); const path = require('path');
const { analyzeSource, CLOSURE, ROOT_PATTERNS, APPROVED_USES } = require('./compute-baseline-hash.cjs');
const dir = path.resolve(process.argv[2]);
const rows = []; const computedRows = []; const allText = {};
for (const f of CLOSURE) {
  const src = fs.readFileSync(path.join(dir, f), 'utf8'); allText[f] = src.split('\n');
  const r = analyzeSource(f, src, CLOSURE.slice());
  for (const x of r.roots) rows.push({ f, ...x });
  for (const c of r.computed) computedRows.push({ f, ...c });
}
// occurrences of the text "globalThis" anywhere (including comments and strings) so nothing is missed by a token-only view
const textual = [];
for (const f of CLOSURE) allText[f].forEach((l, i) => { if (/globalThis/.test(l)) textual.push({ f, line: i + 1, text: l.trim() }); });
const esc = (s) => s.replace(/\|/g, '\\|');
const out = [];
out.push('# globalThis access inventory (Revision 6.1 candidate, not approved)', '',
  'Generated by `policy-report.cjs` from the eight runtime files that `compute-baseline-hash.cjs` analyzes. A reviewer should re-run it and diff.', '',
  `Textual occurrences of the word "globalThis" in the eight files (including comments and strings): **${textual.length}**. Token-level uses of the sensitive roots (\`globalThis\`, \`vm\`): **${rows.length}**, all approved: **${rows.every((x) => x.approved)}**.`, '',
  '## Every `globalThis` token', '',
  '| File | Line | Expression (source line) | Property | Purpose | Approval | Restriction |', '|---|---|---|---|---|---|---|');
for (const x of rows.filter((x) => x.root === 'globalThis')) {
  const p = ROOT_PATTERNS[x.f].find((q) => q.id === x.pattern);
  out.push(`| ${x.f} | ${x.line} | \`${esc(allText[x.f][x.line - 1].trim())}\` | ${p.property} | ${esc(p.purpose)} | ${x.approved ? 'Approved' : 'REJECTED'} | exact token pattern ${p.id} \`${p.seq}\` in ${x.f}, at most once; any other shape, property, alias, computed or optional access, or a second occurrence is refused |`);
}
out.push('', '## Every `vm` token (the other sensitive root)', '', '| File | Line | Expression (source line) | Purpose | Approval | Restriction |', '|---|---|---|---|---|---|');
for (const x of rows.filter((x) => x.root === 'vm')) {
  const p = ROOT_PATTERNS[x.f].find((q) => q.id === x.pattern);
  out.push(`| ${x.f} | ${x.line} | \`${esc(allText[x.f][x.line - 1].trim())}\` | ${esc(p.purpose)} | ${x.approved ? 'Approved' : 'REJECTED'} | exact token pattern ${p.id} \`${p.seq}\`, at most once; \`vm\` may be required only by sdk-loader.cjs |`);
}
out.push('', '## Textual mentions (for completeness)', '', '| File | Line | Text |', '|---|---|---|');
for (const t of textual) out.push(`| ${t.f} | ${t.line} | \`${esc(t.text)}\` |`);
out.push('', '## Approved allowlist (the complete set)', '', '| Id | File | Exact tokens | Property | Max uses |', '|---|---|---|---|---|');
for (const [f, ps] of Object.entries(ROOT_PATTERNS)) for (const p of ps) out.push(`| ${p.id} | ${f} | \`${p.seq}\` | ${p.property} | 1 |`);
out.push('', 'Properties of `globalThis` that are approved: `self` (write) and `StellarSdk` (read), only in sdk-loader.cjs and only in the shapes above. Everything else, including `globalThis.process`, `.require`, `.module`, `.Buffer`, `.eval`, `.Function`, is refused by the pattern rule; no such use exists in the files.', '',
  '## Computed member accesses in the eight files', '', 'Computed access is allowed only with a number, one identifier or simple arithmetic as the key. The files contain:', '', '| File | Line | Key |', '|---|---|---|');
for (const c of computedRows) out.push(`| ${c.f} | ${c.line} | \`${c.key}\` |`);
out.push('', 'None uses a string, template, call or nested bracket as a key, and none is applied to `globalThis`, `process`, `vm`, `module` or `require` (those roots are only accepted in the exact patterns above).', '',
  '## Watched API names and reviewed maximum uses (R5)', '', '```json', JSON.stringify(APPROVED_USES, null, 2), '```', '');
process.stdout.write(out.join('\n'));
