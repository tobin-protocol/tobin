// Revision 6.3 CANDIDATE (NOT APPROVED): mutation check of the AST policy against the eight real runtime files.
// Usage: node mutation-check.cjs <signer dir> [--json out.json]
// Every mutant has an id, source fixture, transformation, expected result, actual result and killed/survived status.
// Expected: every mutant EXCEPT the declared controls must be refused; every control must be accepted.
const fs = require('fs');
const path = require('path');
const { analyze, loadParser } = require('./ast-policy.cjs');

const dir = process.argv[2];
const jsonOut = process.argv.includes('--json') ? process.argv[process.argv.indexOf('--json') + 1] : null;
const FILES = ['authorization.cjs', 'dpapi-key.cjs', 'pinned.cjs', 'refusal.cjs', 'sdk-loader.cjs', 'state-store.cjs', 'tst-signer-core.cjs', 'tst-signer.cjs'];
const P = loadParser();
const mutants = [];
let seq = 0;
const add = (group, file, transformation, source, expected) => mutants.push({ id: `M${String(++seq).padStart(4, '0')}`, group, file, transformation, source, expected });

// ---- baseline: every real file must pass, otherwise nothing below means anything
const base = {};
for (const f of FILES) { base[f] = fs.readFileSync(path.join(dir, f), 'utf8'); if (!analyze(base[f], f).ok) { console.error('BASELINE FAILS for', f); process.exit(2); } }

// ---- Group A: rewrite each real capability call site in place
const MODS = { fs: 1, nodeFs: 1, path: 1, crypto: 1, readline: 1, vm: 1 };
const A_OPS = [
  ['computed literal member', (o, p) => `${o}['${p}']`, 'refuse'],
  ['computed env-derived member', (o) => `${o}[process.env.TOBIN_M]`, 'refuse'],
  ['comma-sequence indirect call', (o, p, T) => `(0, ${T})`, 'refuse'],
  ['optional chaining', (o, p) => `${o}?.${p}`, 'refuse'],
  ['Reflect.get extraction', (o, p) => `Reflect.get(${o}, '${p}')`, 'refuse'],
  ['arrow wrapper returning the member', (o, p, T) => `(() => ${T})()`, 'refuse'],
  ['array storage', (o, p, T) => `[${T}][0]`, 'refuse'],
  ['object storage', (o, p, T) => `({ f: ${T} }).f`, 'refuse'],
  ['.call indirection', (o, p, T) => `${T}.call`, 'refuse'],
  ['logical-or duplicate', (o, p, T) => `(${T} || ${T})`, 'refuse'],
  ['function wrapper returning the member', (o, p, T) => `(function () { return ${T}; })()`, 'refuse'],
  ['.bind indirection', (o, p, T) => `${T}.bind(null)`, 'refuse'],
  ['conditional selection', (o, p, T) => `(process.env.TOBIN_X ? ${T} : ${T})`, 'refuse'],
  ['parenthesised callee (semantically identical; CONTROL)', (o, p, T) => `(${T})`, 'accept'],
];
for (const f of FILES) {
  const ast = P.parse(base[f], { ecmaVersion: 2022, sourceType: 'script', allowReturnOutsideFunction: true, allowHashBang: true });
  const sites = [];
  (function w(n) {
    if (!n || typeof n.type !== 'string') return;
    if (n.type === 'CallExpression' && n.callee.type === 'MemberExpression' && !n.callee.computed && n.callee.object.type === 'Identifier' && MODS[n.callee.object.name]) {
      sites.push({ s: n.callee.start, e: n.callee.end, o: n.callee.object.name, p: n.callee.property.name, T: base[f].slice(n.callee.start, n.callee.end), kind: 'module member call' });
    }
    if (n.type === 'CallExpression' && n.callee.type === 'Identifier' && (n.callee.name === 'exec' || n.callee.name === 'execFileSync')) {
      sites.push({ s: n.callee.start, e: n.callee.end, o: n.callee.name, p: '', T: n.callee.name, kind: 'process-execution call', exec: true });
    }
    for (const k of Object.keys(n)) { const v = n[k]; if (Array.isArray(v)) v.forEach(w); else if (v && typeof v.type === 'string') w(v); }
  })(ast);
  for (const st of sites) {
    for (const [name, fn, expected] of A_OPS) {
      if (st.exec && !/wrapper|array|object|sequence|Reflect|logical|conditional|bind|\.call|parenthesised/.test(name) ) continue;
      if (st.exec && /Reflect|computed|optional/.test(name)) continue;
      let rep;
      if (st.exec) {
        const T = st.T;
        rep = ({ 'comma-sequence indirect call': `(0, ${T})`, 'arrow wrapper returning the member': `(() => ${T})()`, 'array storage': `[${T}][0]`, 'object storage': `({ f: ${T} }).f`,
          '.call indirection': `${T}.call`, 'logical-or duplicate': `(${T} || ${T})`, 'function wrapper returning the member': `(function () { return ${T}; })()`, '.bind indirection': `${T}.bind(null)`,
          'conditional selection': `(process.env.TOBIN_X ? ${T} : ${T})`, 'parenthesised callee (semantically identical; CONTROL)': `(${T})` })[name];
        if (!rep) continue;
      } else rep = fn(st.o, st.p, st.T);
      add('A', f, `${st.kind} at offset ${st.s} (${st.T}): ${name}`, base[f].slice(0, st.s) + rep + base[f].slice(st.e), expected);
    }
  }
}

// ---- Group B: introduce a sensitive reference of each kind into each file (appended)

  }
}

// ---- run
const results = [];
const t0 = Date.now();
for (const m of mutants) {
  let actual, rules = '';
  try { const r = analyze(m.source, m.file); actual = r.ok ? 'accept' : 'refuse'; rules = [...new Set(r.violations.map((v) => v.rule))].sort().join(','); }
  catch (e) { actual = 'refuse'; rules = 'EXC:' + e.message.slice(0, 40); }
  const killed = actual === m.expected;
  results.push({ id: m.id, group: m.group, file: m.file, transformation: m.transformation, expected: m.expected, actual, rules, status: killed ? (m.expected === 'accept' ? 'control-ok' : 'killed') : 'SURVIVED' });
}
const by = (g, s) => results.filter((r) => r.group === g && r.status === s).length;
const lines = [];
lines.push('Revision 6.3 CANDIDATE mutation check (NOT APPROVED). Parser: acorn 8.19.0 (pinned). Seconds: ' + ((Date.now() - t0) / 1000).toFixed(1));
lines.push('group\tmutants\tkilled(refused)\tcontrols-ok\tsurvived');
for (const g of ['A', 'B', 'C', 'D', 'E']) lines.push(`${g}\t${results.filter((r) => r.group === g).length}\t${by(g, 'killed')}\t${by(g, 'control-ok')}\t${by(g, 'SURVIVED')}`);
const surv = results.filter((r) => r.status === 'SURVIVED');
lines.push(`TOTAL ${results.length}; killed ${results.filter((r) => r.status === 'killed').length}; controls accepted ${results.filter((r) => r.status === 'control-ok').length}; SURVIVED ${surv.length}`);
for (const s of surv) lines.push(`SURVIVOR ${s.id} [${s.group}] ${s.file}: ${s.transformation} (expected ${s.expected}, got ${s.actual})`);
// rule histogram of kills
const hist = {};
for (const r of results) if (r.status === 'killed') for (const x of r.rules.split(',')) hist[x] = (hist[x] || 0) + 1;
lines.push('refusals by rule (a mutant may trip several): ' + Object.entries(hist).sort().map(([k, v]) => `${k}=${v}`).join(' '));
console.log(lines.join('\n'));
if (jsonOut) fs.writeFileSync(jsonOut, JSON.stringify({ summary: lines, mutants: results }, null, 1));
process.exit(surv.length ? 1 : 0);
