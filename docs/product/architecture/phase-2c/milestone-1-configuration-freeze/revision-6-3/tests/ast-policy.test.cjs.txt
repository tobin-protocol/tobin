const test = require('node:test');

const SIGNER = process.env.TOBIN_SIGNER_DIR || path.join(__dirname, 'in');
const FILES = ['authorization.cjs', 'dpapi-key.cjs', 'pinned.cjs', 'refusal.cjs', 'sdk-loader.cjs', 'state-store.cjs', 'tst-signer-core.cjs', 'tst-signer.cjs'];
const src = (f) => fs.readFileSync(path.join(SIGNER, f), 'utf8');
const rules = (r) => [...new Set(r.violations.map((v) => v.rule))].sort().join(',');
const append = (file, code) => analyze(src(file) + '\n' + code + '\n', file);
function refused(file, code, expect) {
  const r = append(file, code);
  assert.equal(r.ok, false, `ACCEPTED (must be refused): ${code}`);
  if (expect) assert.ok(expect.split('|').some((x) => r.violations.some((v) => v.rule === x)), `expected one of ${expect}, got ${rules(r)} for: ${code}`);
  return r;
}

// ---------------------------------------------------------------- the demonstrated Revision 6.2 bypass
test('6.3 the exact Revision 6.2 bypass (computed method extraction from child_process) is refused', () => {
  const code = "'use strict';\n\nconst cp = require('child_process');\nconst method = process.env.TOBIN_METHOD;\nconst run = cp[method];\n\nrun('powershell.exe', ['-NoProfile']);\n";
  for (const f of FILES) {
    const r = analyze(code, f);
    assert.equal(r.ok, false, f);
    assert.ok(r.violations.some((v) => v.rule === 'K1'), `${f}: computed extraction refused`);
    assert.ok(r.violations.some((v) => v.rule === 'R1'), `${f}: require form refused`);
  }
  const r = analyze(code, 'state-store.cjs');
  assert.ok(r.violations.some((v) => v.rule === 'K1' && /capability "cp"/.test(v.msg)), 'computed access on the module binding is refused on its own');
});

});

test('6.3 harmless control: whitespace, comments, parenthesised callee, extra pure function and bounded loop keys stay accepted', () => {
  assert.equal(append(SS, "// self and eval mentioned in a comment only\nconst harmless = (a, b) => a + b;\nconst bounded = {};\nfor (const k of Object.keys(bounded)) { void bounded[k]; }\nfor (let i = 0; i < 2; i++) { void [1, 2][i]; }\nconst t = `x ${harmless(1, 2)} y`;").ok, true);
